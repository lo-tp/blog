{{- /* list.html — home, sections, technique pages, and the full technique index. Home is the edition's title page: the masthead, then the spread — the plan on the left (techniques with counts, the year spine, the author), the contents ledger on the right. A technique page is the same ledger with one technique marked, in the author's own spelling of it, not Hugo's lowercased taxonomy key. /tags/ is the index on its own, every technique counted. */ -}} {{- define "main" -}} {{- $records := .Scratch.Get "records" -}} {{- $techs := .Scratch.Get "techniques" -}} {{- $singles := .Scratch.Get "singles" -}} {{- $index := $techs | append $singles -}} {{- if .IsHome -}} {{- /* The header — the masthead band: the edition's name at Part Title scale with `Blog` beside it, then the profile links and the reader's two edition controls — is printed by layouts/partials/header.html, which the shell renders on every page. The title page wears that band exactly as a record does; nothing here repeats or enlarges it. */ -}}
{{- partial "index-rail.html" (dict "page" .) -}} {{- partial "ledger.html" (dict "page" . "pages" $records "paginate" true) -}}
{{- else if eq .Kind "taxonomy" -}} {{- /* The index on its own. The header band above already names it: this block prints no heading and opens no rule, because a page is headed once. */ -}}
    {{- range $entry := $index -}}
  • {{ $entry.name }} {{- with $entry.count -}} {{ . }} {{- end -}}
    • {{- end -}}
{{- else -}} {{- /* A term or section page. Hugo's term title is the lowercased key, so the heading and the index mark both come from the index data instead. */ -}} {{- $key := urlize .Title -}} {{- $currentKey := "" -}} {{- $label := .Title -}} {{- if eq .Kind "term" -}} {{- $currentKey = $key -}} {{- range $entry := $index -}} {{- if eq $entry.key $key -}} {{- $label = $entry.name -}} {{- end -}} {{- end -}} {{- end -}} {{- $pages := where .Pages "Draft" "eq" false -}}
{{- partial "index-rail.html" (dict "page" . "current" $currentKey) -}} {{- partial "ledger.html" (dict "page" . "pages" $pages "paginate" false) -}}
{{- end -}} {{- end -}}
